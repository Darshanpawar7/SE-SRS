import React, { useState } from 'react';
import { buildSprintReport, dailyReportCsv, reportRange, reportTasks, rollingRange } from '../../utils/sprintReports';
import { PriorityBadge, StatusBadge } from '../workspace/TaskSignals';
import SignalIcon from '../workspace/SignalIcon';
import FlowAnalytics from './FlowAnalytics';
import './reportsWorkbench.css';
 
function CompletionRing({ report }) {
  const length = 2 * Math.PI * 62;
  return <article className="report-card report-health"><span className="report-eyebrow">CURRENT SPRINT SNAPSHOT</span><h3>Story point completion</h3><div className="completion-ring"><svg viewBox="0 0 160 160" role="img" aria-label={report.completion === null ? 'No estimated points in this sprint' : `${report.completion}% of current story points completed`}><circle cx="80" cy="80" r="62" className="ring-track"/><circle cx="80" cy="80" r="62" className="ring-value" strokeDasharray={length} strokeDashoffset={length * (1 - (report.completion || 0) / 100)} style={{ '--ring-length': length }}/></svg><strong>{report.completion === null ? '—' : `${report.completion}%`}<small>{report.donePoints} / {report.points} points</small></strong></div><p>Completed points divided by the points currently in this sprint. Scope changes can change this value.</p></article>;
}
function StatusDonut({ report, choose, selection }) {
  const length = 2 * Math.PI * 52, total = report.statuses.reduce((n, s) => n + s.count, 0);
  const segments = report.statuses.filter(s => s.count);
  return <article className="report-card"><span className="report-eyebrow">CURRENT WORK ITEMS</span><h3>Workflow distribution</h3><div className="status-donut"><svg viewBox="0 0 160 160" role="img" aria-label={`${total} work items by current status`}><circle cx="80" cy="80" r="52" className="ring-track"/>{segments.map((s, index) => {
    const segment = s.count / total * length, start = segments.slice(0,index).reduce((n,v) => n + v.count / total * length, 0);
    return <circle key={s.id} cx="80" cy="80" r="52" fill="none" stroke={s.color} strokeWidth="16" strokeDasharray={`${segment} ${length-segment}`} strokeDashoffset={-start} transform="rotate(-90 80 80)"><title>{s.label}: {s.count}</title></circle>;
  })}<text x="80" y="80" textAnchor="middle" dominantBaseline="middle">{total}</text><text x="80" y="100" textAnchor="middle" className="donut-caption">work items</text></svg><div className="report-legend">{report.statuses.map(s => <button key={s.id} onClick={() => choose('status',s.id,s.label)} aria-pressed={selection?.kind==='status'&&selection.id===s.id}><i style={{background:s.color}}/><span>{s.label}</span><b>{s.count}</b></button>)}</div></div></article>;
}
export default function ReportsWorkbench({ tasks, activities, project, sprint, onOpenTask }) {
  const [dates, setDates] = useState(() => rollingRange());
  const [selection, setSelection] = useState(null);
  const range = reportRange(dates.from, dates.to);
  const report = buildSprintReport(tasks, activities, range);
  const choose = (kind,id,label) => setSelection(old => old?.kind===kind&&old.id===id ? null : {kind,id,label});
  const visible = report ? reportTasks(tasks,report,selection) : [];
  const download = () => {
    const url = URL.createObjectURL(new Blob([dailyReportCsv(report)],{type:'text/csv;charset=utf-8'}));
    const anchor = document.createElement('a');anchor.href=url;anchor.download=`SprintFlow-completions-${range.from}-${range.to}.csv`;anchor.click();setTimeout(()=>URL.revokeObjectURL(url),1000);
  };
  const maxOwner = Math.max(1,...(report?.owners.map(o=>o.points)||[]));
  const maxPriority = Math.max(1,...(report?.priorities.map(p=>p.count)||[]));
  const firstWeekday = range.error ? 0 : (new Date(`${range.from}T00:00:00Z`).getUTCDay()+6)%7;
  return <section className="reports-workbench" aria-labelledby="reports-heading"><header className="report-toolbar"><div><span className="report-eyebrow">DELIVERY INTELLIGENCE</span><h3 id="reports-heading">Sprint reports</h3><p>{sprint?.name || 'No sprint selected'} · UTC date ranges · recorded data</p></div><div className="report-presets" aria-label="Report duration">{[7,28,90].map(days=><button key={days} className="btn btn-secondary btn-sm" onClick={()=>{setDates(rollingRange(days));setSelection(null);}}>{days} days</button>)}</div></header>
    <div className="report-date-controls"><label>From (UTC)<input type="date" className="form-input" value={dates.from} onChange={e=>{setDates({...dates,from:e.target.value});setSelection(null);}}/></label><label>To (UTC)<input type="date" className="form-input" value={dates.to} onChange={e=>{setDates({...dates,to:e.target.value});setSelection(null);}}/></label><button className="btn btn-secondary" disabled={!report} onClick={download}><SignalIcon name="export" size={15}/>Export daily counts</button><p>Dates filter completion history. Snapshot charts use all current items in the selected sprint.</p></div>
    {range.error ? <p className="form-error" role="alert">{range.error}</p> : <>
      <div className="report-grid" key={`${range.from}/${range.to}`}><CompletionRing report={report}/><StatusDonut report={report} choose={choose} selection={selection}/>
      <article className="report-card report-heatmap"><span className="report-eyebrow">{report.completedTransitions} RECORDED COMPLETION TRANSITIONS</span><h3>Completion activity</h3><p>{range.from} — {range.to}. Click a date to inspect its work items.</p><div className="heatmap-scroll"><div className="heatmap-days">{['Mon','Tue','Wed','Thu','Fri','Sat','Sun'].map(d=><span key={d}>{d}</span>)}</div><div className="heatmap-grid" style={{gridTemplateColumns:`repeat(${Math.ceil((firstWeekday+range.days)/7)},minmax(16px,1fr))`}}>{report.throughput.map((d,i)=><button key={d.date} className={`heatmap-cell level-${Math.min(4,d.count)}`} style={{gridColumn:Math.floor((firstWeekday+i)/7)+1,gridRow:(firstWeekday+i)%7+1}} title={`${d.date}: ${d.count} recorded completions`} aria-label={`${d.date}: ${d.count} recorded completions`} aria-pressed={selection?.kind==='date'&&selection.id===d.date} onClick={()=>choose('date',d.date,d.date)}/>)}</div></div><p className="report-footnote">A zero means no retained completion event. Reopened tasks can complete again; this is not a productivity score.</p></article>
      <article className="report-card"><span className="report-eyebrow">CURRENT PRIORITY × STATUS</span><h3>Priority breakdown</h3><div className="report-priorities">{report.priorities.map(p=><button key={p.id} aria-pressed={selection?.kind==='priority'&&selection.id===p.id} onClick={()=>choose('priority',p.id,p.label)}><span>{p.label}<b>{p.count} items</b></span><i className="report-stack" style={{width:`${Math.max(2,p.count/maxPriority*100)}%`}}>{p.states.map(s=><i key={s.id} style={{background:s.color,width:`${p.count?s.count/p.count*100:0}%`}}/>)}</i></button>)}</div><details><summary>Priority and status values</summary><div className="report-table-scroll"><table><thead><tr><th>Priority</th>{report.statuses.map(s=><th key={s.id}>{s.label}</th>)}</tr></thead><tbody>{report.priorities.map(p=><tr key={p.id}><th>{p.label}</th>{p.states.map(s=><td key={s.id}>{s.count}</td>)}</tr>)}</tbody></table></div></details></article>
      <article className="report-card"><span className="report-eyebrow">CURRENT ASSIGNEES · STORY POINTS</span><h3>Workload distribution</h3><p>Assigned estimates show work distribution, not individual performance.</p><div className="report-workload">{report.owners.map(o=><button key={o.id} aria-pressed={selection?.kind==='owner'&&selection.id===o.id} onClick={()=>choose('owner',o.id,o.name)}><span>{o.name}<b>{o.points} pt · {o.openPoints} open</b></span><i className="owner-track"><i style={{width:`${o.points/maxOwner*100}%`,background:o.color}}/></i></button>)}</div></article>
      <article className="report-card"><span className="report-eyebrow">CURRENT OPEN WORK</span><h3>Oldest work items</h3><p>Age since a valid creation timestamp, measured now.</p><div className="report-oldest">{report.ages.slice(0,5).map(({task,days})=><button key={task.id} onClick={()=>onOpenTask(task)}><span>{task.title}</span><b>{Math.floor(days)} d</b></button>)}{!report.ages.length&&<p>No open work with recorded creation dates.</p>}</div></article></div>
      <section className="report-card report-drilldown" aria-labelledby="report-drilldown-heading"><header><div><span className="report-eyebrow">CHART DRILL-DOWN</span><h3 id="report-drilldown-heading">{selection?`${selection.label} · ${visible.length} matching items`:`All ${visible.length} sprint items`}</h3></div>{selection&&<button className="btn btn-secondary btn-sm" onClick={()=>setSelection(null)}>Clear chart selection</button>}</header><div className="report-table-scroll"><table><thead><tr><th>Work item</th><th>Status</th><th>Priority</th><th>Points</th></tr></thead><tbody>{visible.map(t=><tr key={t.id}><td><button className="task-title-link" onClick={()=>onOpenTask(t)}>{t.title}</button></td><td><StatusBadge status={t.status}/></td><td><PriorityBadge priority={t.priority}/></td><td>{t.story_points}</td></tr>)}</tbody></table></div>{!visible.length&&<p className="report-footnote">No current sprint items match this selection.</p>}</section>
      <FlowAnalytics project={project} sprint={sprint} tasks={tasks} activities={activities} onOpenTask={onOpenTask} range={range}/>
      <p className="report-footnote">History covers retained audit events for items currently in this sprint. Deleted or moved items are excluded. Chart selections open the matching current tasks; no missing history is reconstructed.</p>
    </>}
  </section>;
}
